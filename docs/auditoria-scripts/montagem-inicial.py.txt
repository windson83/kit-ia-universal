from pathlib import Path, PurePosixPath
import hashlib, io, json, re, zipfile

BASE = Path(__file__).resolve().parent
OUT = BASE / 'KIT-IA-UNIVERSAL'
OUT.mkdir(exist_ok=True)
def write(rel, value):
    p = OUT / rel
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(value.rstrip() + '\n', encoding='utf-8')

# Inventory includes archives nested inside archives; nothing is executed/extracted to arbitrary paths.
entries, blobs = [], {}
def scan(label, data, depth=0):
    item = {'origem': label, 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}
    entries.append(item)
    if label.lower().endswith(('.zip', '.skill')):
        if depth > 5: raise ValueError('Archive nesting limit')
        with zipfile.ZipFile(io.BytesIO(data)) as z:
            for f in z.infolist():
                if f.is_dir(): continue
                path = PurePosixPath(f.filename.replace('\\', '/'))
                if path.is_absolute() or '..' in path.parts or ':' in str(path):
                    raise ValueError('Unsafe archive member: ' + f.filename)
                if f.file_size > 20_000_000: raise ValueError('Oversized member')
                scan(label + '!/' + str(path), z.read(f), depth + 1)
    else:
        try:
            txt = data.decode('utf-8-sig')
            blobs[label] = txt
            item['linhas'] = len(txt.splitlines())
            item['tipo'] = 'script' if label.endswith(('.py','.sh','.ps1')) else 'documento'
        except UnicodeDecodeError:
            item['tipo'] = 'binario'

for p in sorted(BASE.iterdir()):
    if p.name in ('KIT-IA-UNIVERSAL', 'montar_distribuicao.py'): continue
    if p.is_file(): scan(p.name, p.read_bytes())
    elif p.name == 'agents':
        for f in sorted(p.rglob('*')):
            if f.is_file(): scan(f.relative_to(BASE).as_posix(), f.read_bytes())
write('docs/inventario-legado.json', json.dumps(entries, ensure_ascii=False, indent=2))

## Limites

A auditoria é estática e os testes validam a distribuição e o instalador. Não é certificação do comportamento de um modelo nem teste de acesso ao SEI, serviços externos ou execução de pipelines reais. Referências técnicas recuperadas são material de apoio histórico e devem ser confrontadas com documentação atual quando versões ou recursos importarem. Não foi feita análise jurídica de um caso concreto nem certificação de todas as normas citadas nos originais. Snapshots em origens/ são histórico, não instruções ativas.
